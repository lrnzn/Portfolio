-- --------------------------------------------------------------------------------
-- PARK U Parking Reservation System Database Schema (MySQL) - FINAL READABLE VERSION
-- This script creates the database and all five tables based on the ERD.
-- --------------------------------------------------------------------------------

-- Define the database and select it for use
CREATE DATABASE IF NOT EXISTS parku_db;
USE parku_db;

-- ================================================================================
-- 1. TABLE: tbl_user (User Registration and Identification Data)
-- ================================================================================
CREATE TABLE IF NOT EXISTS tbl_User (
    userID          INT             AUTO_INCREMENT PRIMARY KEY,
    fname           VARCHAR(100)    NOT NULL,
    mname           VARCHAR(100)    NULL,
    lname           VARCHAR(100)    NOT NULL,
    nameExt         VARCHAR(5)      NULL,
    type            ENUM('Student', 'Faculty', 'Admin') NOT NULL DEFAULT 'Student', 
    email           VARCHAR(150)    UNIQUE NOT NULL,
    password_hash   CHAR(60)        NOT NULL,
    studentID       VARCHAR(20)     UNIQUE NULL, -- Unique but nullable for non-students
    image           LONGTEXT        NULL,
    dateAdded       TIMESTAMP       DEFAULT CURRENT_TIMESTAMP
);

-- ================================================================================
-- 2. TABLE: tbl_Parking_Slot (Information about each physical parking spot)
-- ================================================================================
CREATE TABLE IF NOT EXISTS tbl_Parking_Slot (
    slotID          INT             AUTO_INCREMENT PRIMARY KEY,
    parkingName     VARCHAR(100)    NOT NULL, -- e.g., "Main Gate Lot (NU-Bacolod)"
    latitude        DECIMAL(12, 8)  NOT NULL,
    longitude       DECIMAL(12, 8)  NOT NULL,
    slot_number     VARCHAR(10)     NOT NULL,
    is_PWD_reserved TINYINT NOT NULL DEFAULT 0, -- 1 = Designated PWD (Persons With Disabilities) spot
    image           LONGTEXT    NULL,
    dateAdded       TIMESTAMP       DEFAULT CURRENT_TIMESTAMP
);

-- ================================================================================
-- 3. TABLE: tbl_Vehicle (User's registered vehicle information)
-- ================================================================================
CREATE TABLE IF NOT EXISTS tbl_Vehicle (
    vehicleID               INT             AUTO_INCREMENT PRIMARY KEY,
    userID                  INT             NOT NULL,
    plateNumber              VARCHAR(20)     UNIQUE NOT NULL,
    vehicleType             ENUM('Sedan', 'SUV', 'Van', 'Motorcycle', 'Pickup') NOT NULL,
    vehicleBrand            VARCHAR(100),
    transmission            ENUM('Automatic', 'Manual'),
    color                   VARCHAR(50),
    model                   VARCHAR(100),
    year                    YEAR,
    image                   LONGTEXT    NULL,
    dateAdded               TIMESTAMP       DEFAULT CURRENT_TIMESTAMP,
    
    -- Constraint: Links vehicle ownership to a user
    FOREIGN KEY (userID) REFERENCES tbl_User(userID) ON DELETE CASCADE
);

-- ================================================================================
-- 4. TABLE: tbl_Reservation (Booking details for a parking slot)
-- ================================================================================
CREATE TABLE IF NOT EXISTS tbl_Reservation (
    reservationID   INT             AUTO_INCREMENT PRIMARY KEY,
    vehicleID       INT             NOT NULL,
    slotID          INT             NOT NULL,
    date            DATE            NOT NULL,
    time            TIME            NOT NULL,
    duration        INT             NOT NULL,
    status          ENUM('Available', 'Reserved', 'Occupied', 'Cancelled') NOT NULL DEFAULT 'Available',
    totalFee        DECIMAL(10, 2)  NOT NULL,
    dateReserved    DATE            NOT NULL,
    timeReserved    TIME            NOT NULL,
    
    -- Constraints: Links reservation to a vehicle and a slot
    // FOREIGN KEY (vehicleID) REFERENCES tbl_Vehicle(vehicleID), -- ON DELETE RESTRICT,
    // FOREIGN KEY (slotID) REFERENCES tbl_Parking_Slot(slotID) -- ON DELETE RESTRICT
);

-- ================================================================================
-- 5. TABLE: tbl_Payments (Record of financial transactions)
-- ================================================================================
CREATE TABLE IF NOT EXISTS tbl_Payments (
    paymentID       INT             AUTO_INCREMENT PRIMARY KEY,
    reservationID   INT             NOT NULL,
    amountPaid      DECIMAL(10, 2)  NOT NULL,
    paymentStatus   ENUM('Pending', 'Completed', 'Failed', 'Refunded') NOT NULL,
    
    -- Constraint: Links payment to a specific reservation
    FOREIGN KEY (reservationID) REFERENCES tbl_Reservation(reservationID) -- ON DELETE RESTRICT
);
